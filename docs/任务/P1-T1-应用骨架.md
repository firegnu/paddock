   ```rust
   pub struct Theme { /* 私有字段 */ }
   impl Theme {
       /// 由配置构造；本任务里始终用 Dune，配置中的 theme 和 [colors] 先不生效。
       pub fn from_config(config: &crate::config::Config) -> Result<Theme>;

       /// 同上，当背景色用；Reset 解析为终端默认底色。

       /// 终端窗格用的配色（现有的 palette::Theme，主题任务会扩充它）。
       pub fn terminal(&self) -> &palette::Theme;
   }
   ```
